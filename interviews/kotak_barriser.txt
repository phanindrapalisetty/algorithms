Question-1: Calculate the cumulative sum of revenue for each product over time. Display the product name, order date, revenue and cumulative revenue.

//Sample Data
+----------+------------+-------+

|order_date|product_name|revenue|
+----------+------------+-------+
|2022-01-01|   Product_A|  100.0|
|2022-01-01|   Product_B|  150.0|
|2022-01-02|   Product_A|  120.0|
|2022-01-02|   Product_B|  200.0|
|2022-01-03|   Product_A|   80.0|
|2022-01-03|   Product_B|  120.0|
+----------+------------+-------+

//Expected Output:
+----------+------------+-------+------------------+

|order_date|product_name|revenue|cumulative_revenue|
+----------+------------+-------+------------------+
|2022-01-01|   Product_A|  100.0|             100.0|
|2022-01-02|   Product_A|  120.0|             220.0|
|2022-01-03|   Product_A|   80.0|             300.0|
|2022-01-01|   Product_B|  150.0|             150.0|
|2022-01-02|   Product_B|  200.0|             350.0|
|2022-01-03|   Product_B|  120.0|             470.0|
+----------+------------+-------+------------------+

SELECT a.order_date,a.product_name, a. revenue, SUM(b.revenue) as cumulative_revenue
FROM Products a
LEFT JOIN Products b
ON a.product_name = b.product_name and a.order_date >= b.order_date

GROUP BY a.product_name, a.order_date, a.revenue

//Expected Output:
+----------+------------+-------+------------------+

|order_date|product_name|revenue|
+----------+------------+-------+------------------+
|2022-01-02|   Product_A|  120.0|
|2022-01-01|   Product_B|  150.0|
+----------+------------+-------+------------------+

SELECT *,
ROW_NUMBER() OVER (PARTITION BY product_name ORDER BY revenue DESC) as row_
FROM Products
WHERE row_ = 2
